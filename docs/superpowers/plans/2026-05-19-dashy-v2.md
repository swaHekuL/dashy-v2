# Dashy v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `dashy-v2`, a passive auto-rotating Next.js dashboard for a Raspberry Pi 3B+ (800×480) that shows a persistent clock header and cycles through 5 data panels every 7 seconds.

**Architecture:** Pure Next.js (no Python). Next.js API routes handle all external data fetching with in-memory caching. The frontend polls its own API routes and drives panel rotation with `setInterval`. Google OAuth refresh token is stored in a gitignored `config/credentials.json`.

**Tech Stack:** Next.js (pages router), React, `googleapis` npm package (Calendar + Gmail OAuth), `rss-parser` npm package (News), OpenWeatherMap REST API (Weather), Steam Store public API (Steam), Bebas Neue + Roboto via `next/font/google`.

## Task 2: Google OAuth script

**Files:**
- Create: `dashy-v2/scripts/google-auth.js`

Before writing this script you need a Google Cloud project with Calendar API and Gmail API enabled, plus OAuth 2.0 credentials of type "Desktop app". The redirect URI `http://127.0.0.1:3333/callback` must be added as an authorized redirect URI in the Google Cloud Console.

Add `client_id` and `client_secret` from those credentials into `config/credentials.json` before running the script.

- [ ] **Step 1: Create `scripts/google-auth.js`**

```js
const { google } = require('googleapis');
const http = require('http');
const { URL } = require('url');
const fs = require('fs');
const path = require('path');

const CREDS_PATH = path.join(__dirname, '../config/credentials.json');
const PORT = 3333;

async function main() {
  const creds = JSON.parse(fs.readFileSync(CREDS_PATH, 'utf8'));
  const oauth2Client = new google.auth.OAuth2(
    creds.client_id,
    creds.client_secret,
    `http://127.0.0.1:${PORT}/callback`
  );

  const authUrl = oauth2Client.generateAuthUrl({
    access_type: 'offline',
    scope: [
      'https://www.googleapis.com/auth/calendar.readonly',
      'https://www.googleapis.com/auth/gmail.readonly',
    ],
    prompt: 'consent',
  });

  console.log('\nOpen this URL in your browser:\n\n' + authUrl + '\n');

  await new Promise((resolve, reject) => {
    const server = http.createServer(async (req, res) => {
      const parsed = new URL(req.url, `http://127.0.0.1:${PORT}`);
      const code = parsed.searchParams.get('code');
      if (!code) { res.end('No code in request.'); return; }
      res.end('Auth complete — you can close this tab.');
      server.close();
      try {
        const { tokens } = await oauth2Client.getToken(code);
        creds.refresh_token = tokens.refresh_token;
        fs.writeFileSync(CREDS_PATH, JSON.stringify(creds, null, 2));
        console.log('\nRefresh token saved to config/credentials.json');
        resolve();
      } catch (e) {
        reject(e);
      }
    });
    server.listen(PORT, '127.0.0.1', () => {
      console.log(`Waiting for OAuth callback on http://127.0.0.1:${PORT}/callback ...`);
    });
  });
}

main().catch(err => { console.error(err); process.exit(1); });
```

```js
import { useState, useEffect } from 'react';
import Clock from '../screens/Clock';

const PANELS = ['weather', 'calendar', 'gmail', 'news', 'steam'];
const PANEL_MS = 7000;
const REFRESH_MS = {
  weather:  10 * 60 * 1000,
  calendar:  5 * 60 * 1000,
  gmail:     2 * 60 * 1000,
  news:     15 * 60 * 1000,
  steam:    60 * 60 * 1000,
};

export default function Home() {
  const [panelIndex, setPanelIndex] = useState(0);
  const [data, setData] = useState({
    weather: null, calendar: null, gmail: null, news: null, steam: null,
  });

  const fetchPanel = async (panel) => {
    try {
      const res = await fetch(`/api/${panel}`);
      if (!res.ok) return;
      setData(prev => ({ ...prev, [panel]: await res.json() }));
    } catch {}
  };

  useEffect(() => {
    PANELS.forEach(p => fetchPanel(p));
    const intervals = PANELS.map(p => setInterval(() => fetchPanel(p), REFRESH_MS[p]));
    return () => intervals.forEach(clearInterval);
  }, []);

  useEffect(() => {
    const id = setInterval(() => setPanelIndex(i => (i + 1) % PANELS.length), PANEL_MS);
    return () => clearInterval(id);
  }, []);

  const current = PANELS[panelIndex];

  return (
    <div style={{ width: '100vw', height: '100vh', overflow: 'hidden', display: 'flex', flexDirection: 'column' }}>
      <Clock />
      <div style={{ flex: 1, overflow: 'hidden', background: '#000', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
        <span style={{ color: '#333', fontFamily: 'monospace', fontSize: '2vw', letterSpacing: '0.2em' }}>
          {current.toUpperCase()}
        </span>
      </div>
    </div>
  );
}
```

```js
const creds = require('../../config/credentials.json');

let cache = null;
let cacheAt = 0;
const TTL = 10 * 60 * 1000;

export default async function handler(req, res) {
  const now = Date.now();
  if (cache && now - cacheAt < TTL) return res.json(cache);

  try {
    const r = await fetch(
      `https://api.openweathermap.org/data/2.5/weather?lat=${creds.owm_lat}&lon=${creds.owm_lon}&units=imperial&appid=${creds.owm_api_key}`
    );
    if (!r.ok) throw new Error(`OWM ${r.status}`);
    const d = await r.json();

    const fr = await fetch(
      `https://api.openweathermap.org/data/2.5/forecast?lat=${creds.owm_lat}&lon=${creds.owm_lon}&units=imperial&cnt=8&appid=${creds.owm_api_key}`
    );
    const fd = fr.ok ? await fr.json() : null;
    const temps = fd ? fd.list.map(i => i.main.temp) : [];

    cache = {
      temp: Math.round(d.main.temp),
      condition: d.weather[0].description,
      humidity: d.main.humidity,
      wind: Math.round(d.wind.speed),
      windDir: d.wind.deg,
      high: temps.length ? Math.round(Math.max(...temps)) : null,
      low: temps.length ? Math.round(Math.min(...temps)) : null,
      rain: Math.round((d.rain?.['1h'] ?? 0) * 10) / 10,
    };
    cacheAt = now;
    res.json(cache);
  } catch {
    if (cache) return res.json(cache);
    res.status(503).json({ error: 'unavailable' });
  }
}
```

```js
import { google } from 'googleapis';
const creds = require('../../config/credentials.json');

let cache = null;
let cacheAt = 0;
const TTL = 5 * 60 * 1000;

function getAuth() {
  const auth = new google.auth.OAuth2(creds.client_id, creds.client_secret);
  auth.setCredentials({ refresh_token: creds.refresh_token });
  return auth;
}

export default async function handler(req, res) {
  const now = Date.now();
  if (cache && now - cacheAt < TTL) return res.json(cache);

  try {
    const calendar = google.calendar({ version: 'v3', auth: getAuth() });
    const start = new Date();
    start.setHours(0, 0, 0, 0);
    const end = new Date();
    end.setHours(23, 59, 59, 999);

    const r = await calendar.events.list({
      calendarId: 'primary',
      timeMin: start.toISOString(),
      timeMax: end.toISOString(),
      singleEvents: true,
      orderBy: 'startTime',
      maxResults: 3,
    });

    cache = {
      events: (r.data.items || []).map(e => ({
        title: e.summary || '(no title)',
        time: e.start.dateTime
          ? new Date(e.start.dateTime).toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit' })
          : 'All day',
        color: e.colorId ? COLOR_MAP[e.colorId] : '#1a73e8',
      })),
    };
    cacheAt = now;
    res.json(cache);
  } catch {
    if (cache) return res.json(cache);
    res.status(503).json({ error: 'unavailable' });
  }
}

const COLOR_MAP = {
  '1': '#7986cb', '2': '#33b679', '3': '#8e24aa', '4': '#e67c73',
  '5': '#f6c026', '6': '#f5511d', '7': '#039be5', '8': '#616161',
```

```js
import { google } from 'googleapis';
const creds = require('../../config/credentials.json');

let cache = null;
let cacheAt = 0;
const TTL = 2 * 60 * 1000;

function getAuth() {
  const auth = new google.auth.OAuth2(creds.client_id, creds.client_secret);
  auth.setCredentials({ refresh_token: creds.refresh_token });
  return auth;
}

export default async function handler(req, res) {
  const now = Date.now();
  if (cache && now - cacheAt < TTL) return res.json(cache);

  try {
    const gmail = google.gmail({ version: 'v1', auth: getAuth() });

    const listRes = await gmail.users.messages.list({
      userId: 'me',
      labelIds: ['UNREAD', 'INBOX'],
      maxResults: 3,
    });

    const unreadCount = listRes.data.resultSizeEstimate || 0;
    const messages = listRes.data.messages || [];

    const previews = await Promise.all(
      messages.slice(0, 3).map(async ({ id }) => {
        const msg = await gmail.users.messages.get({
          userId: 'me',
          id,
          format: 'metadata',
          metadataHeaders: ['Subject', 'From'],
        });
        const headers = msg.data.payload.headers;
        const subject = headers.find(h => h.name === 'Subject')?.value ?? '(no subject)';
        const from = headers.find(h => h.name === 'From')?.value ?? '';
        const sender = from.replace(/<.*>/, '').trim() || from;
        return { subject, sender };
      })
    );

    cache = { unreadCount, previews };
    cacheAt = now;
    res.json(cache);
  } catch {
    if (cache) return res.json(cache);
    res.status(503).json({ error: 'unavailable' });
  }
}
```

```js
import Parser from 'rss-parser';

let cache = null;
let cacheAt = 0;
const TTL = 15 * 60 * 1000;
const RSS_URL = 'https://feeds.bbci.co.uk/news/rss.xml';

    cacheAt = now;
```

## Task 8: Steam API route + panel

**Files:**
- Create: `dashy-v2/pages/api/steam.js`
- Create: `dashy-v2/screens/Steam.jsx`

Uses Steam's public `store.steampowered.com/api/featured/` endpoint — no auth required. Also fetches `api/featuredcategories/` to get new releases.

- [ ] **Step 1: Create `pages/api/steam.js`**

```js
let cache = null;
let cacheAt = 0;
const TTL = 60 * 60 * 1000;

export default async function handler(req, res) {
  const now = Date.now();
  if (cache && now - cacheAt < TTL) return res.json(cache);

  try {
    const [featRes, catRes] = await Promise.all([
      fetch('https://store.steampowered.com/api/featured/?cc=us&l=en'),
      fetch('https://store.steampowered.com/api/featuredcategories/?cc=us&l=en'),
    ]);

    const feat = featRes.ok ? await featRes.json() : null;
    const cat = catRes.ok ? await catRes.json() : null;

    const deals = [];
    const allFeatured = [
      ...(feat?.large_capsules || []),
      ...(feat?.featured_win || []),
    ];
    for (const item of allFeatured) {
      if (item.discount_percent > 0 && deals.length < 3) {
        deals.push({
          name: item.name,
          discount: item.discount_percent,
        });
      }
    }

    const newRelease = cat?.new_releases?.items?.[0]?.name ?? null;

    cache = { deals, newRelease };
    cacheAt = now;
    res.json(cache);
  } catch {
    if (cache) return res.json(cache);
    res.status(503).json({ error: 'unavailable' });
  }
}
```
